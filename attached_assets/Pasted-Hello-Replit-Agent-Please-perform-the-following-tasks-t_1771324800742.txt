Task 3: Refactor the App for Multilingual SEO

This task involves creating separate components for each language and setting up routing and dynamic meta tags.

Step 1: Create a Seo.tsx component

Create a new file at src/components/Seo.tsx and add the following code. This component will manage all SEO-related tags for each page.

TypeScript


import React from 'react';
import { Helmet } from 'react-helmet-async';

interface SeoProps {
  lang: 'pt' | 'en' | 'es';
  title: string;
  description: string;
  keywords: string;
  canonicalUrl: string;
}

const Seo: React.FC<SeoProps> = ({ lang, title, description, keywords, canonicalUrl } ) => {
  const domain = 'https://tours.lisbonwineroutes.com';

  return (
    <Helmet>
      <html lang={lang} />
      <title>{title}</title>
      <meta name="description" content={description} />
      <meta name="keywords" content={keywords} />
      <link rel="canonical" href={canonicalUrl} />

      {/* Hreflang Tags */}
      <link rel="alternate" hreflang="pt" href={`${domain}/`} />
      <link rel="alternate" hreflang="en" href={`${domain}/en/`} />
      <link rel="alternate" hreflang="es" href={`${domain}/es/`} />
      <link rel="alternate" hreflang="x-default" href={`${domain}/en/`} />

      {/* Open Graph */}
      <meta property="og:title" content={title} />
      <meta property="og:description" content={description} />
      <meta property="og:url" content={canonicalUrl} />
      <meta property="og:type" content="website" />
      <meta property="og:image" content={`${domain}/og-image.jpg`} />
      <meta property="og:locale" content={lang === 'pt' ? 'pt_PT' : lang === 'en' ? 'en_US' : 'es_ES'} />

      {/* Twitter Card */}
      <meta name="twitter:card" content="summary_large_image" />
      <meta name="twitter:title" content={title} />
      <meta name="twitter:description" content={description} />
      <meta name="twitter:image" content={`${domain}/twitter-card.jpg`} />

      {/* Schema.org JSON-LD */}
      <script type="application/ld+json">{`
        {
          "@context": "https://schema.org",
          "@type": "Service",
          "name": "Lisbon Wine Routes - Personalized Wine Itineraries",
          "description": "Create your personalized wine tourism itinerary in Lisbon with AI-powered curation.",
          "url": "${domain}",
          "provider": {
            "@type": "Organization",
            "name": "Lisbon Wine Routes",
            "url": "https://www.lisbonwineroutes.com"
          },
          "areaServed": {
            "@type": "City",
            "name": "Lisboa"
          },
          "serviceType": "Wine Tourism Planning"
        }
      `}</script>
    </Helmet>
   );
};

export default Seo;

File: src/App.tsx

Set up the routes for each language. Note that English (/en/) is now the default route.

TypeScript


import React from 'react';
import { BrowserRouter as Router, Routes, Route, Navigate } from 'react-router-dom';
import HomePagePT from './pages/HomePagePT';
import HomePageEN from './pages/HomePageEN';
import HomePageES from './pages/HomePageES';

function App() {
  return (
    <Router>
      <Routes>
        <Route path="/" element={<HomePagePT />} />
        <Route path="/en" element={<HomePageEN />} />
        <Route path="/es" element={<HomePageES />} />
        {/* Redirect any other path to the default English page */}
        <Route path="*" element={<Navigate to="/en" replace />} />
      </Routes>
    </Router>
  );
}

export default App;
